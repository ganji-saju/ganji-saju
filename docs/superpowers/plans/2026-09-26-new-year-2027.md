```ts
export function buildNewYearScopeKey(readingKey: string, year: number) {
  return `newyear:${readingKey}:${year}`;
}

export function parseNewYearScopeKey(
  scopeKey: string | null | undefined
): { readingKey: string; year: number } | null {
  const trimmed = scopeKey?.trim() ?? '';
  if (!trimmed.startsWith('newyear:')) return null;
  const parts = trimmed.split(':');
  if (parts.length < 3) return null;
  const year = parseYearScope(parts[parts.length - 1]);
  if (year === null) return null;
  const readingKey = parts.slice(1, -1).join(':');
  return readingKey ? { readingKey, year } : null;
}
```

```ts
export async function hasNewYearEntitlementForReading(
  userId: string | null | undefined,
  readingKey: string | null | undefined,
  year: number
): Promise<boolean> {
  if (!userId || !readingKey || !hasSupabaseServiceEnv) return false;
  const currentIdentity = sajuIdentityFromReadingKey(readingKey);
  const scopeKeys = await listTasteProductEntitlementScopeKeys(userId, 'new-year');
  return scopeKeys.some((scopeKey) => {
    const parsed = parseNewYearScopeKey(scopeKey);
    return (
      parsed !== null &&
      parsed.year === year &&
      readingKeyMatchesCurrentSaju(parsed.readingKey, [readingKey], currentIdentity)
    );
  });
}
```

```ts
//   청구액은 반드시 applyCouponDiscount 로 계산한다 — resolveChargeForUser 와 createPaymentOrder 가 같은 함수를 써야 화면가 = order.amount.
import { applyCouponDiscount } from '@/lib/coupons/discount-coupon';

export const MEMBER_DISCOUNT_PERCENT_BY_PACKAGE: Readonly<Record<string, number>> = {
  taste_new_year_2027: 50,
};

export function pickBetterDiscount(
  listAmount: number,
  memberPercent: number,
  coupon: { percent: number; maxDiscountWon?: number | null } | null
): 'member' | 'coupon' | 'none' {
  const member = memberPercent > 0 ? applyCouponDiscount(listAmount, memberPercent, null).discountWon : 0;
  const byCoupon = coupon ? applyCouponDiscount(listAmount, coupon.percent, coupon.maxDiscountWon ?? null).discountWon : 0;
  if (member === 0 && byCoupon === 0) return coupon ? 'coupon' : 'none';
  return member >= byCoupon ? 'member' : 'coupon';
}
```

```ts
export function parseNewYearExtrasText(text: string, fallback: SajuNewYearExtras) {
  const fail = (msg: string) => ({ ok: false as const, extras: fallback, errorMessage: msg });
  try {
    const p = JSON.parse(extractJsonCandidate(text)) as Record<string, unknown>;
    const cats = (p.categories ?? {}) as Record<string, unknown>;
    const family = cleanText(cats.family, MAX_CATEGORY_LENGTH);
    const study = cleanText(cats.study, MAX_CATEGORY_LENGTH);
    const isCat = (v: unknown): v is NewYearHighlightCategory => NEW_YEAR_HIGHLIGHT_CATEGORIES.includes(v as NewYearHighlightCategory);
    const quarters = Array.isArray(p.quarterlyFlows) ? p.quarterlyFlows : [];
    const quarterlyFlows = QUARTER_MONTHS.map((months, i) => {
      const row = quarters.find((q) => (q as { quarter?: unknown })?.quarter === i + 1) as Record<string, unknown> | undefined;
      const summary = cleanText(row?.summary, MAX_PERIOD_LENGTH);
      return summary ? { quarter: (i + 1) as 1|2|3|4, months, summary, focusCategory: isCat(row?.focusCategory) ? row!.focusCategory : 'work' } : null;
    });
    const highlights = (v: unknown) => (Array.isArray(v) ? v : []).map((h) => {
      const r = h as Record<string, unknown>;
      const month = typeof r.month === 'number' ? r.month : Number.parseInt(String(r.month), 10);
      const t = cleanText(r.text, MAX_PERIOD_LENGTH);
      return Number.isInteger(month) && month >= 1 && month <= 12 && isCat(r.category) && t ? { month, category: r.category, text: t } : null;
    }).filter((h): h is NewYearHighlight => h !== null).slice(0, 6);
    const expectations = highlights(p.expectations);
    const cautions = highlights(p.cautions);
    if (!family || !study || quarterlyFlows.some((q) => !q) || expectations.length < 3 || cautions.length < 3) {
      return fail('New-year extras JSON is missing required sections.');
    }
    return { ok: true as const, extras: { categories: { family, study }, quarterlyFlows: quarterlyFlows as NewYearQuarterFlow[], expectations, cautions }, errorMessage: null };
  } catch (e) {
    return fail(e instanceof Error ? e.message : 'New-year extras JSON could not be parsed.');
  }
}
```
